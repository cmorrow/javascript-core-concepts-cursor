Challenge: Add a Reset Button

Task: Add a reset button that sets count to 0 and text to "Waiting...".

Waiting...

Solution Code

// Find the reset button and add a click listener
document.querySelector('#reset').addEventListener('click', () => {
    // Reset the count variable to 0
    count = 0;
    
    // Find the output paragraph
    const out = document.querySelector('#out');
    
    // Reset the text content to "Waiting..."
    out.textContent = 'Waiting...';
    
    // Reset the color back to black (or original color)
    out.style.color = 'black';
});

Solution 3: Create Reset Function

// Create a reusable reset function
function resetCounter() {
    count = 0;
    const out = document.querySelector('#out');
    out.textContent = 'Waiting...';
    out.style.color = 'black';
}

// Use the function
document.querySelector('#reset').addEventListener('click', resetCounter);
