Video 7: Async JavaScript - Challenge Solution

Challenge: Fetch Random Dog Image

Task: Fetch a random dog image from https://dog.ceo/api/breeds/image/random and show it in an <img id="dog"> element.

Solution Code

// Find the button and add click listener
document.querySelector('#fetch').addEventListener('click', async () => {
    // Find the img element
    const img = document.querySelector('#dog');
    
    // Clear previous image (optional)
    img.src = '';
    img.style.display = 'none';
    
    try {
        // Fetch data from the API
        const res = await fetch('https://dog.ceo/api/breeds/image/random');
        
        // Convert response to JavaScript object
        const data = await res.json();
        
        // Set the image source to the URL from the API
        // data.message contains the image URL
        img.src = data.message;
        img.style.display = 'block';  // Show the image
    } catch {
        // If something goes wrong, show error
        img.alt = 'Failed to load dog 😢';
        img.style.display = 'block';
    }
});

Alternative Solutions

Solution 1: With Status Message

document.querySelector('#fetch').addEventListener('click', async () => {
    const img = document.querySelector('#dog');
    const status = document.querySelector('#status');
    
    status.textContent = 'Loading...';
    img.style.display = 'none';
    
    try {
        const res = await fetch('https://dog.ceo/api/breeds/image/random');
        const data = await res.json();
        
        img.src = data.message;
        img.style.display = 'block';
        status.textContent = 'Dog loaded!';
    } catch (error) {
        status.textContent = 'Failed to load dog 😢';
        img.style.display = 'none';
    }
});

Solution 2: With Error Handling

document.querySelector('#fetch').addEventListener('click', async () => {
    const img = document.querySelector('#dog');
    
    try {
        img.src = '';  // Clear previous image
        
        const res = await fetch('https://dog.ceo/api/breeds/image/random');
        
        // Check if the response is OK
        if (!res.ok) {
            throw new Error('API returned an error');
        }
        
        const data = await res.json();
        
        // Check if data has the expected structure
        if (!data.message) {
            throw new Error('Invalid data format');
        }
        
        img.src = data.message;
        img.alt = 'Random Dog';
    } catch (error) {
        img.alt = 'Failed to load dog 😢';
        console.error('Error:', error);
    }
});

Solution 3: Using .then() Instead of async/await

// This is the older way (still works, but async/await is cleaner)
document.querySelector('#fetch').addEventListener('click', () => {
    const img = document.querySelector('#dog');
    
    img.src = '';
    
    fetch('https://dog.ceo/api/breeds/image/random')
        .then(res => res.json())
        .then(data => {
            img.src = data.message;
        })
        .catch(() => {
            img.alt = 'Failed to load dog 😢';
        });
});
